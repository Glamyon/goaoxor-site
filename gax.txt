<script>
  // 顶部公告栏消息轮播数组
  const messages = [
    'We take on fewer projects to keep coordination clear.',
    'Matching is done manually, not automatically.',
    'Availability varies depending on project scope.'
  ];
  let msgIndex = 0;
  
  // FAQ toggle
  document.querySelectorAll('.faq-item').forEach(item => {
    const question = item.querySelector('.faq-question');
    const toggle = item.querySelector('.faq-toggle');
    
    question.addEventListener('click', () => {
      const isActive = item.classList.contains('active');
      
      // Close all other FAQ items
      document.querySelectorAll('.faq-item').forEach(otherItem => {
        if (otherItem !== item) {
          otherItem.classList.remove('active');
        }
      });
      
      // Toggle current item
      item.classList.toggle('active');
      
      // Update toggle symbol
      toggle.textContent = item.classList.contains('active') ? '−' : '+';
    });
  });

  // Modal functionality
  const modal = document.getElementById('modalOverlay');
  const modalClose = document.getElementById('modalClose');
  const remindLater = document.getElementById('remindLater');
  
  // Show modal after delay
  setTimeout(() => {
    modal.classList.add('active');
  }, 1500);
  
  // Close modal on X button
  modalClose.addEventListener('click', () => {
    modal.classList.remove('active');
  });
  
  // Close modal on "I'll come back later"
  remindLater.addEventListener('click', () => {
    modal.classList.remove('active');
  });
  
  // Close modal when clicking outside
  modal.addEventListener('click', (e) => {
    if (e.target === modal) {
      modal.classList.remove('active');
    }
  });
  
  // Close modal with Escape key
  document.addEventListener('keydown', (e) => {
    if (e.key === 'Escape' && modal.classList.contains('active')) {
      modal.classList.remove('active');
    }
  });

  // Top bar functionality
  const topBar = document.getElementById('topBar');
  const closeTopBar = document.getElementById('closeTopBar');
  
  // Close top bar
  closeTopBar.addEventListener('click', () => {
    topBar.style.transform = 'translateY(-100%)';
    setTimeout(() => {
      topBar.style.display = 'none';
    }, 300);
  });
  
  // Rotate top bar messages
  function rotateMessage() {
    topBar.style.opacity = 0;
    setTimeout(() => {
      // 移除旧的关闭按钮
      const oldBtn = topBar.querySelector('.close-top-bar');
      if (oldBtn) oldBtn.remove();
      
      // 更新消息文本
      const textNode = Array.from(topBar.childNodes).find(node => node.nodeType === 3);
      if (textNode) {
        textNode.textContent = ' ' + messages[msgIndex];
      } else {
        topBar.textContent = messages[msgIndex];
      }
      
      // 重新添加关闭按钮
      const newCloseBtn = document.createElement('button');
      newCloseBtn.className = 'close-top-bar';
      newCloseBtn.textContent = '×';
      newCloseBtn.id = 'closeTopBar';
      newCloseBtn.addEventListener('click', () => {
        topBar.style.transform = 'translateY(-100%)';
        setTimeout(() => {
          topBar.style.display = 'none';
        }, 300);
      });
      topBar.appendChild(newCloseBtn);
      
      topBar.style.opacity = 1;
      msgIndex = (msgIndex + 1) % messages.length;
    }, 300);
  }
  
  setInterval(rotateMessage, 4000);
  
  // Add scroll animations
  const observerOptions = {
    root: null,
    rootMargin: '0px',
    threshold: 0.1
  };
  
  const observer = new IntersectionObserver((entries) => {
    entries.forEach(entry => {
      if (entry.isIntersecting) {
        entry.target.classList.add('slide-up');
      }
    });
  }, observerOptions);
  
  // Observe all elements with the slide-up class
  document.querySelectorAll('.slide-up').forEach(el => {
    observer.observe(el);
  });
</script>
